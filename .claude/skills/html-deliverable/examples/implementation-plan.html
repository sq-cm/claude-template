Transform Lambda — field-mapping snippet (high risk: normalisation logic) 
// transform/mapContact.js
// RISK: legacy records may carry null email — guard before HubSpot upsert
export function mapContact(rawRecord) {
  if (!rawRecord?.email) {
    throw new TransformError('Missing email', { rawRecord });
  }
  return {
    properties: {
      email:      rawRecord.email.toLowerCase().trim(),
      firstname:  rawRecord.first_name ?? '',
      lastname:   rawRecord.last_name  ?? '',
      phone:      rawRecord.mobile ?? rawRecord.phone ?? '',
      // SQ custom property: original platform ID
      sq_platform_id: String(rawRecord.id),
    }
  };
}
Load Lambda — HubSpot rate-limit back-off (high risk) 
// load/hubspotClient.js
// HubSpot API: 100 req/10s burst, 40,000 req/day
// Implement exponential back-off on 429 responses
const MAX_RETRIES = 4;

export async function upsertContact(payload, attempt = 0) {
  const res = await fetch('https://api.hubapi.com/crm/v3/objects/contacts/upsert', {
    method: 'POST',
    headers: { 'Authorization': `Bearer ${process.env.HS_ACCESS_TOKEN}`,
               'Content-Type': 'application/json' },
    body: JSON.stringify(payload),
  });

  if (res.status === 429 && attempt < MAX_RETRIES) {
    const delay = Math.pow(2, attempt) * 500;
    await new Promise(r => setTimeout(r, delay));
    return upsertContact(payload, attempt + 1);
  }

  if (!res.ok) throw new LoadError(await res.text(), { status: res.status });
  return res.json();
}

Infrastructure

All resources provisioned via Terraform (module: sq-crm-pipeline). State stored in S3 with DynamoDB lock table. Region: ap-southeast-2. Environments: staging, prod.

  • EventBridge rule triggers Ingest Lambda on e-commerce order events.
  • SQS FIFO queue between Ingest and Transform — deduplication window 5 minutes.
  • Dead-letter queue retention: 14 days. CloudWatch alarm on DLQ depth > 0.
  • Secrets Manager holds HS_ACCESS_TOKEN — rotated every 90 days.
  • Lambda concurrency limits: Ingest 50, Transform 20, Load 10 (HubSpot rate ceiling).
